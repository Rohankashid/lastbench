2. Step-by-Step Quadratic Equation Algorithm

  • Step 1: Start.
  • Step 2: Read floating-point coefficients a, b, c.
  • Step 3: If a == 0, print "Degenerate equation: Linear (x = -c/b)" and terminate.
  • Step 4: Compute discriminant: D = (b * b) - (4 * a * c).
  • Step 5: If D > 0: Roots are real and distinct. Compute r1 = (-b + sqrt(D)) / (2*a) and r2 = (-b - sqrt(D)) / (2*a).
  • Step 6: Else If D == 0: Roots are real and equal. Compute r1 = r2 = -b / (2*a).
  • Step 7: Else (D < 0): Roots are complex conjugates. Compute real = -b / (2*a) and imag = sqrt(-D) / (2*a). Display real ± i*imag.
  • Step 8: Stop.

auto: Allocated on Stack frame. Local scope. Destroyed on block exit. Default: Garbage. register: Stored in CPU register (or stack fallback). Fast access, no &address allowed. static: Retained in BSS/Data segment throughout execution. Initialized once. Default: 0. extern: Stored in Data segment with external global linkage across all source files. 
#include <stdio.h>

void executeScopeDemo() {
    auto int autoCounter = 1;     /* Re-created on stack upon each invocation */
    static int staticCounter = 1; /* Initialized ONCE in data segment; value persists */
    printf("autoCounter = %d | staticCounter = %d\n", autoCounter, staticCounter);
    autoCounter++;
    staticCounter++;
}

int main() {
    printf("Call 1: "); executeScopeDemo(); /* autoCounter = 1 | staticCounter = 1 */
    printf("Call 2: "); executeScopeDemo(); /* autoCounter = 1 | staticCounter = 2 */
    printf("Call 3: "); executeScopeDemo(); /* autoCounter = 1 | staticCounter = 3 */
    return 0;
}

Q3. Explain Bitwise Operators in C with truth tables (&, |, ^, ~, <<, >>). For operands a = 12 and b = 25, compute: a & b, a | b, a ^ b, a << 2, and b >> 2 with binary arithmetic steps. SPPU May 2019, Dec 2023, May 2024 • 7 Marks

Operand binary representations in 8-bit notation: a = 12 = 00001100_2, b = 25 = 00011001_2.

  • Bitwise AND (a & b): 00001100 & 00011001 = 00001000_2 = 8_10
  • Bitwise OR (a | b): 00001100 | 00011001 = 00011101_2 = 29_10
  • Bitwise XOR (a ^ b): 00001100 ^ 00011001 = 00010101_2 = 21_10
  • Left Shift (a << 2): 00001100 << 2 = 00110000_2 = 48_10 (Formula: 12 * 2^2 = 48)
  • Right Shift (b >> 2): 00011001 >> 2 = 00000110_2 = 6_10 (Formula: floor(25 / 2^2) = 6)

Q4. Compare while, do-while, and for loops across 5 structural criteria. Write a complete C program to check whether a given integer is an Armstrong Number with a step-by-step dry-run trace table for input 153. SPPU Dec 2019, May 2023, Dec 2024 • 8 Marks 
#include <stdio.h>

int main() {
    int num, originalNum, remainder, accumulatedSum = 0;
    printf("Enter a 3-digit positive integer: ");
    if (scanf("%d", &num) != 1 || num < 0) return 1;
    
    originalNum = num;
    while (originalNum != 0) {
        remainder = originalNum % 10;                         /* Extract terminal digit */
        accumulatedSum += (remainder * remainder * remainder); /* Cube and accumulate */
        originalNum /= 10;                                    /* Truncate terminal digit */
    }
    
    if (accumulatedSum == num)
        printf("%d is an Armstrong Number.\n", num);
    else
        printf("%d is NOT an Armstrong Number.\n", num);
    return 0;
}

Q5. Write a complete C program to perform Matrix Multiplication of two matrices A (r1 x c1) and B (r2 x c2). State the mandatory dimension compatibility condition and explain the triple-nested loop architecture. SPPU Dec 2022, May 2023, May 2024 • 8 Marks 
#include <stdio.h>
#define MAX 10

int main() {
    int a[MAX][MAX], b[MAX][MAX], result[MAX][MAX];
    int r1, c1, r2, c2, i, j, k;
    printf("Enter Matrix A rows and columns: "); scanf("%d %d", &r1, &c1);
    printf("Enter Matrix B rows and columns: "); scanf("%d %d", &r2, &c2);
    
    /* Dimensional Compatibility Check */
    if (c1 != r2) {
        printf("Error: Matrix multiplication undefined (cols of A != rows of B).\n");
        return 1;
    }
    
    /* Input Matrix A and B omitted for brevity */
    for (i = 0; i < r1; i++) {
        for (j = 0; j < c2; j++) {
            result[i][j] = 0; /* Reset dot product accumulator */
            for (k = 0; k < c1; k++) {
                result[i][j] += a[i][k] * b[k][j];
            }
        }
    }
    return 0;
}

Q6. Differentiate Call by Value versus Call by Reference (Address) with stack memory diagrams. Write two C functions demonstrating why variable swapping fails in Call by Value and succeeds in Call by Reference. SPPU Dec 2019, May 2022, May 2024 • 8 Marks

#include <stdio.h>

/* FAILS: Call by Value (Modifies local stack copies only) */
void swapByValue(int x, int y) {
    int temp = x; x = y; y = temp;
}

/* SUCCEEDS: Call by Reference (Dereferences caller memory addresses) */
void swapByReference(int *x, int *y) {
    int temp = *x; *x = *y; *y = temp;
}

int main() {
    int a = 10, b = 20;
    swapByValue(a, b);
    printf("After Value: a = %d, b = %d (UNALTERED)\n", a, b);
    swapByReference(&a, &b);
    printf("After Reference: a = %d, b = %d (SWAPPED)\n", a, b);
    return 0;
}

#include <stdio.h>

struct Student {
    int rollNo;
    char name[32];
    float marks[3];
    float percentage;
};

int main() {
    struct Student students[5]; int topperIndex = 0;
    for (int i = 0; i < 5; i++) {
        printf("\nEnter Record for Student %d:\n", i + 1);
        scanf("%d %s", &students[i].rollNo, students[i].name);
        float total = 0;
        for (int j = 0; j < 3; j++) {
            scanf("%f", &students[i].marks[j]);
            total += students[i].marks[j];
        }
        students[i].percentage = total / 3.0f;
        if (students[i].percentage > students[topperIndex].percentage) topperIndex = i;
    }
    printf("\nTOPPER: Roll: %d, Name: %s, Percentage: %.2f%%\n",
           students[topperIndex].rollNo, students[topperIndex].name, students[topperIndex].percentage);
    return 0;
}
